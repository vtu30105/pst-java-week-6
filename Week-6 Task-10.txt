import java.util.*;

class Solution {

    static int longestPalindrome(String s) {
        int n = s.length();
        int[] d1 = new int[n];
        int[] d2 = new int[n];

        // Odd length palindromes
        for (int i = 0, l = 0, r = -1; i < n; i++) {
            int k = (i > r) ? 1 : Math.min(d1[l + r - i], r - i + 1);

            while (i - k >= 0 && i + k < n &&
                   s.charAt(i - k) == s.charAt(i + k))
                k++;

            d1[i] = k--;

            if (i + k > r) {
                l = i - k;
                r = i + k;
            }
        }

        // Even length palindromes
        for (int i = 0, l = 0, r = -1; i < n; i++) {
            int k = (i > r) ? 0 : Math.min(d2[l + r - i + 1], r - i + 1);

            while (i - k - 1 >= 0 && i + k < n &&
                   s.charAt(i - k - 1) == s.charAt(i + k))
                k++;

            d2[i] = k--;

            if (i + k > r) {
                l = i - k - 1;
                r = i + k;
            }
        }

        int ans = 1;

        for (int x : d1)
            ans = Math.max(ans, 2 * x - 1);

        for (int x : d2)
            ans = Math.max(ans, 2 * x);

        return ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        String s = sc.next();

        for (int k = 0; k < n; k++) {

            // Create rotation
            String rotation = s.substring(k) + s.substring(0, k);

            System.out.println(longestPalindrome(rotation));
        }

        sc.close();
    }
}
